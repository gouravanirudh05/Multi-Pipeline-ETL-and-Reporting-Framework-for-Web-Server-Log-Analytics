TABLE etl_runs
  run_id           SERIAL PK
  pipeline         VARCHAR(20)    -- pig | mapreduce | mongodb | hive
  run_uuid         VARCHAR(64)    UNIQUE
  batch_size       INTEGER
  total_records    INTEGER
  total_batches    INTEGER
  avg_batch_size   NUMERIC(10,2)
  malformed_count  INTEGER
  runtime_seconds  NUMERIC(10,3)
  status           VARCHAR(20)    -- running | completed | failed
  started_at       TIMESTAMPTZ
  completed_at     TIMESTAMPTZ

TABLE daily_traffic
  id               SERIAL PK
  run_id           INTEGER FK → etl_runs
  pipeline         VARCHAR(20)
  run_uuid         VARCHAR(64)
  batch_id         INTEGER
  executed_at      TIMESTAMPTZ
  log_date         DATE
  status_code      INTEGER
  request_count    BIGINT
  total_bytes      BIGINT

TABLE q2_top_resources
  id                    SERIAL PK
  run_id                INTEGER FK → etl_runs
  pipeline              VARCHAR(20)
  run_uuid              VARCHAR(64)
  batch_id              INTEGER
  executed_at           TIMESTAMPTZ
  resource_path         TEXT
  request_count         BIGINT
  total_bytes           BIGINT
  distinct_host_count   BIGINT

TABLE q3_hourly_errors
  id                    SERIAL PK
  run_id                INTEGER FK → etl_runs
  pipeline              VARCHAR(20)
  run_uuid              VARCHAR(64)
  batch_id              INTEGER
  executed_at           TIMESTAMPTZ
  log_date              DATE
  log_hour              SMALLINT
  error_request_count   BIGINT
  total_request_count   BIGINT
  error_rate            NUMERIC(6,4)
  distinct_error_hosts  BIGINT

-- Indexes for query performance --
idx_q1_pipeline_date    ON q1_daily_traffic(pipeline, log_date)
idx_q2_pipeline         ON q2_top_resources(pipeline, request_count DESC)
idx_q3_pipeline_date    ON q3_hourly_errors(pipeline, log_date, log_hour)
idx_runs_uuid           ON etl_runs(run_uuid)
